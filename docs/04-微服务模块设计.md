### 4.1.3 数据库表设计

```sql
-- 用户表
CREATE TABLE sys_user (
  user_id BIGINT PRIMARY KEY AUTO_INCREMENT,
  username VARCHAR(64) UNIQUE NOT NULL COMMENT '用户名',
  password VARCHAR(255) NOT NULL COMMENT '密码(加密)',
  real_name VARCHAR(64) COMMENT '真实姓名',
  phone VARCHAR(20) UNIQUE COMMENT '手机号',
  email VARCHAR(128) COMMENT '邮箱',
  avatar_url VARCHAR(255) COMMENT '头像URL',
  status TINYINT DEFAULT 1 COMMENT '状态(1正常,0禁用)',
  login_time TIMESTAMP COMMENT '最后登录时间',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  deleted_at TIMESTAMP NULL COMMENT '软删除',
  INDEX idx_username (username),
  INDEX idx_phone (phone),
  INDEX idx_status (status)
);

-- 角色表
CREATE TABLE sys_role (
  role_id BIGINT PRIMARY KEY AUTO_INCREMENT,
  role_name VARCHAR(64) UNIQUE NOT NULL COMMENT '角色名',
  role_code VARCHAR(64) UNIQUE NOT NULL COMMENT '角色代码',
  description VARCHAR(500) COMMENT '描述',
  status TINYINT DEFAULT 1 COMMENT '状态',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX idx_role_code (role_code)
);

-- 权限表
CREATE TABLE sys_permission (
  permission_id BIGINT PRIMARY KEY AUTO_INCREMENT,
  permission_name VARCHAR(64) NOT NULL COMMENT '权限名',
  permission_code VARCHAR(128) UNIQUE NOT NULL COMMENT '权限代码',
  description VARCHAR(500) COMMENT '描述',
  status TINYINT DEFAULT 1,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  INDEX idx_permission_code (permission_code)
);

-- 用户角色关联表
CREATE TABLE sys_user_role (
  user_id BIGINT NOT NULL,
  role_id BIGINT NOT NULL,
  PRIMARY KEY (user_id, role_id),
  FOREIGN KEY (user_id) REFERENCES sys_user(user_id),
  FOREIGN KEY (role_id) REFERENCES sys_role(role_id)
);

-- 角色权限关联表
CREATE TABLE sys_role_permission (
  role_id BIGINT NOT NULL,
  permission_id BIGINT NOT NULL,
  PRIMARY KEY (role_id, permission_id),
  FOREIGN KEY (role_id) REFERENCES sys_role(role_id),
  FOREIGN KEY (permission_id) REFERENCES sys_permission(permission_id)
);
```

CREATE TABLE charging_pile (
  pile_id BIGINT PRIMARY KEY AUTO_INCREMENT,

  pile_number VARCHAR(64) UNIQUE NOT NULL COMMENT '充电桩编号',

  longitude DECIMAL(11, 8) COMMENT '经度',
  address VARCHAR(255) COMMENT '详细地址',

### 4.6.1 Order Service调用其他服务

// Order Service中调用Charging Service
@FeignClient(name = "charging-service")
public interface ChargingServiceClient {
    @GetMapping("/api/charging/{pileId}")
    ChargingPile getChargingPile(@PathVariable Long pileId);
    
    @GetMapping("/api/charging/{pileId}/available-slots")
    Integer getAvailableSlots(@PathVariable Long pileId);
}

// 使用示例
@Service
public class OrderService {
    @Resource
    private ChargingServiceClient chargingServiceClient;
    
    public Order createOrder(Long userId, Long pileId) {
        // 1. 查询充电桩信息
        ChargingPile pile = chargingServiceClient.getChargingPile(pileId);
        
        // 2. 检查是否有可用充电口
        Integer availableSlots = chargingServiceClient.getAvailableSlots(pileId);
        if (availableSlots <= 0) {
            throw new BusinessException("暂无可用充电口");
        }
        
        // 3. 创建订单
        Order order = new Order();
        order.setUserId(userId);
        order.setPileId(pileId);
        // ...
        
        return orderMapper.insert(order);
    }
}
